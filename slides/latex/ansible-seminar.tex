\section{Kiến trúc Agentless}
% =====================================================================

\begin{frame}{Agentless là gì?}
  \begin{itemize}\setlength{\itemsep}{0.4cm}
    \item \textbf{Agent} = chương trình chạy thường trực trên mỗi máy được quản lý, liên lạc với
          server trung tâm để nhận cấu hình (Puppet agent, Chef Infra Client, salt-minion).
    \item \textbf{Agentless} = không cài agent riêng của Ansible trên máy đích. Ansible chỉ cài trên
          \textbf{Control Node}, khi cần thì kết nối tới \textbf{Managed Node} qua kênh có sẵn
          (\textbf{SSH} với Linux, WinRM/SSH với Windows), làm xong thì rút ra.
  \end{itemize}
  \vspace{0.5cm}
  \begin{fitnote}
    Ansible \textbf{chủ động đẩy} (push) cấu hình khi được ra lệnh, thay vì mỗi máy
    phải cài agent và tự kéo (pull) định kỳ.
  \end{fitnote}
\end{frame}

\begin{frame}{Yêu cầu của Control Node và Managed Node}
  \begin{fittable}{>{\bfseries}p{1.8cm} Y Y}
    & \hd{Control Node} & \hd{Managed Node (Linux)} \\
    Ansible & Cài \code{ansible} / \code{ansible-core} & Không cần \\
    Python & Phiên bản được bản Ansible hỗ trợ
           & Cần cho phần lớn module (không bắt buộc cùng phiên bản với Control Node) \\
    SSH & SSH client & SSH server đang chạy, mở cổng TCP 22 \\
    Khác & Inventory, playbook, private key
         & Tài khoản có shell; quyền \code{sudo} nếu task cần root \\
  \end{fittable}
  \vspace{0.3cm}
  \begin{itemize}
    \item Control Node phải là \textbf{Linux/macOS} (Windows thì dùng WSL);
          Managed Node có thể là Windows.
    \item Module \code{raw} chạy được cả khi máy đích chưa có Python
          $\rightarrow$ dùng để cài Python lần đầu.
    \item \code{become: true} = chạy bằng \code{sudo}; \textbf{không tự cấp quyền} sudo
          (\code{-K} để nhập mật khẩu sudo).
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Playbook \& Task}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
\begin{codeblock}[minted options={fontsize=\scriptsize}]{yaml}
# site.yml = Playbook (danh sách play)
- name: Cài nginx          # <- Play
  hosts: webservers
  become: true
  tasks:
    - name: Đảm bảo nginx đã cài  # <- Task
      ansible.builtin.apt:        # module
        name: nginx
        state: present    # trạng thái cần có
\end{codeblock}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{itemize}\small\setlength{\itemsep}{0.22cm}
        \item \textbf{Playbook} = file YAML chứa 1+ \textbf{Play}.
              Play = \code{hosts} + \code{tasks} (chạy từ trên xuống),
              thêm \code{become}, \code{vars}, \code{handlers}.
        \item \textbf{Task} = \code{name} + module + tham số; hay kèm
              \code{when}, \code{loop}, \code{register}, \code{notify}.
        \item \code{state: present} = \textbf{khai báo}, không ra lệnh:
              chưa có $\rightarrow$ cài (\code{changed}), có rồi $\rightarrow$ bỏ qua (\code{ok}).
        \item \code{ansible.builtin.apt} = tên đầy đủ (FQCN). YAML thụt lề bằng \textbf{space}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

        \item Ansible \textbf{tự nạp} \code{main.yml} trong từng thư mục,

        \item Role \textbf{không tự chạy}: playbook gọi bằng \code{roles:}.
        \item Secret để ở \code{group\_vars/}, \textbf{không} nằm trong role
              $\rightarrow$ role \textbf{dùng lại} được ở môi trường khác.
      \end{itemize}
    \end{column}

\begin{frame}[fragile]{Handler}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{itemize}\small\setlength{\itemsep}{0.25cm}
        \item Chỉ chạy khi task \code{notify} báo \textbf{\code{changed}} — config không đổi
              thì không restart.
        \item Chạy \textbf{cuối play}, \textbf{1 lần mỗi host} dù được notify nhiều lần.
        \item Tên trong \code{notify} phải \textbf{khớp} tên handler.
        \item Có task \textbf{lỗi} giữa chừng $\rightarrow$ handler \textbf{bị bỏ qua};
              thêm \code{force\_handlers: true} để vẫn chạy.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
\begin{codeblock}[minted options={fontsize=\scriptsize}]{yaml}
# tasks/main.yml
- name: Deploy cấu hình Nginx
  ansible.builtin.template:
    src: nginx.conf.j2
    dest: /etc/nginx/sites-available/default
  notify: Restart Nginx

# handlers/main.yml
- name: Restart Nginx
  ansible.builtin.service:
    name: nginx
    state: restarted
\end{codeblock}
    \end{column}
  \end{columns}

\section{Idempotence}
% =====================================================================

\begin{frame}{Idempotent là gì?}
  \begin{itemize}\setlength{\itemsep}{0.25cm}
    \item \textbf{Idempotent} (luỹ đẳng): làm \textbf{nhiều lần} cho \textbf{cùng kết quả}
          như làm \textbf{một lần}.
    \item Ansible: chạy playbook \textbf{1 lần hay 100 lần}, máy luôn ở \textbf{cùng trạng thái mong muốn}.
          Lần đầu sửa chỗ còn lệch (\code{changed}), các lần sau chỉ \code{ok}.
    \item Đời thường: \textbf{nút tầng 5} thang máy bấm bao nhiêu lần cũng tới tầng 5 (idempotent);
          \textbf{công tắc đèn} bấm 2 lần lại tắt (không idempotent).
  \end{itemize}
  \vspace{0.25cm}
  \centering
  \begin{tikzpicture}[
      run/.style={rounded corners=3pt, draw=fitBlue, thick, fill=fitLight, align=center,
        font=\scriptsize, minimum width=2.6cm, minimum height=0.85cm},
      ar/.style={-{Stealth[length=2mm]}, thick, fitBlue}]
    \node[run] (r1) {\textbf{Lần 1}\\\code{changed=3}};
    \node[run, right=1cm of r1] (r2) {\textbf{Lần 2}\\\code{changed=0}};
    \node[run, right=1cm of r2] (rn) {\textbf{Lần N}\\\code{changed=0}};
    \draw[ar] (r1) -- (r2); \draw[ar] (r2) -- (rn);
  \end{tikzpicture}
  \par\vspace{0.25cm}
  \begin{fitnote}
    \textbf{Cốt lõi}: chạy lại playbook luôn \textbf{an toàn} — sửa máy lệch cấu hình,
    chạy tiếp sau lỗi, thêm máy mới mà máy cũ không bị ảnh hưởng.
  \end{fitnote}
\end{frame}

\begin{frame}[fragile]{Ansible làm idempotent thế nào?}
  \begin{columns}[c]
    \begin{column}{0.47\textwidth}
      {\small\textbf{Mệnh lệnh} (shell) — \emph{``làm thế nào''}}
\begin{codeblock}[minted options={fontsize=\scriptsize}]{bash}
apt-get install -y nginx
cp nginx.conf /etc/nginx/
systemctl restart nginx  # luôn restart
\end{codeblock}
      \vspace{0.1cm}
      {\small\textbf{Khai báo} (Ansible) — \emph{``kết quả cuối là gì''}}
\begin{codeblock}[minted options={fontsize=\scriptsize}]{yaml}
ansible.builtin.apt:
  name: nginx
  state: present   # có rồi -> ok
\end{codeblock}
      \vspace{0.1cm}
      {\small Module \textbf{kiểm tra hiện trạng} $\rightarrow$ khớp thì \code{ok},
       lệch thì sửa vừa đủ $\rightarrow$ \code{changed}.}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{fittable}{>{\ttfamily\color{fitCode}}l Y}
        \hd{PLAY RECAP} & \hd{Ý nghĩa} \\
        ok & Thành công (gồm cả \code{changed}) \\
        changed & Đã thay đổi máy đích \\
        failed & Lỗi, dừng trên host đó \\
        unreachable & Không SSH được tới host \\
        skipped & \code{when} trả về false \\
      \end{fittable}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Xem trước: \texttt{-{}-check} \& \texttt{-{}-diff}}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}
      \begin{itemize}\small\setlength{\itemsep}{0.25cm}
        \item \code{-{}-check} (\code{-C}): \textbf{mô phỏng} — báo task nào \emph{sẽ}
              \code{changed}, \textbf{không đổi gì} trên máy.
        \item \code{-{}-diff} (\code{-D}): hiện dòng \textbf{thêm (+) / bớt (−)} trong file,
              giống \code{git diff}.
        \item Dùng chung để \textbf{xem trước} khi chạy thật trên production.
        \item Giới hạn: \code{shell}/\code{command} bị \textbf{skip} trong check mode
              (trừ khi có \code{creates}/\code{removes}).
      \end{itemize}
    \end{column}
    \begin{column}{0.54\textwidth}
\begin{codeblock}[minted options={fontsize=\scriptsize}]{diff}
$ ansible-playbook site.yml --check --diff

TASK [nginx : Deploy cấu hình Nginx] ****
--- before: /etc/nginx/sites-available/default
+++ after: nginx.conf.j2
@@ -4 +4 @@
-    server_name demo.ansible.lab;
+    server_name seminar.ansible.lab;
changed: [web-1]
\end{codeblock}
      {\scriptsize\itshape Đổi \code{server\_name} rồi xem trước — file trên máy vẫn chưa đổi.}
    \end{column}
  \end{columns}
\end{frame}

\section{Module chuyên dụng vs Shell}
% =====================================================================

\begin{frame}[fragile]{\texttt{command} vs \texttt{shell}}
  \begin{fittable}{>{\bfseries}p{4.2cm} Y Y}
    & \hd{\code{\color{white}command}} & \hd{\code{\color{white}shell}} \\
    Chạy qua \code{/bin/sh} & Không & Có \\
    Pipe, redirect, \code{\$VAR} & Không & Có \\
    Rủi ro shell injection & \textcolor{fitGreen}{\textbf{Thấp}} & \textcolor{fitRed}{\textbf{Cao}} \\
  \end{fittable}
  \vspace{0.3cm}
  {\small\textbf{Bẫy redirect}: cùng một lệnh, kết quả khác nhau}\par\vspace{-0.2cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
\begin{codeblock}[minted options={fontsize=\scriptsize}]{yaml}
- ansible.builtin.command:
    cmd: echo data >> /tmp/x
# stdout: "data >> /tmp/x"
# KHÔNG tạo file, vẫn changed
\end{codeblock}
    \end{column}
    \begin{column}{0.48\textwidth}
\begin{codeblock}[minted options={fontsize=\scriptsize}]{yaml}
- ansible.builtin.shell:
    cmd: echo data >> /tmp/x
# /bin/sh xử lý >>
# ghi "data" vào /tmp/x
\end{codeblock}
    \end{column}
  \end{columns}
  \vspace{0.35cm}
  {\small Ngoài ra: \code{raw} (không cần Python, dùng để bootstrap) và \code{script}
   (chạy script có sẵn) chỉ dùng trong trường hợp đặc biệt.}
\end{frame}

\begin{frame}{Chạy lại lần 2: shell vs module}
  \begin{fittable}{l Y l Y}
    \hd{Dùng shell} & \hd{Lần 2} & \hd{Dùng module} & \hd{Lần 2} \\
    \code{mkdir /data} & \textcolor{fitRed}{\textbf{lỗi}} {\scriptsize(File exists)}
      & \code{file: state=directory} & \textcolor{fitGreen}{\textbf{ok}} \\
    \code{echo ... >> sysctl.conf} & \textcolor{fitRed}{\textbf{thêm dòng trùng}}
      & \code{lineinfile} & \textcolor{fitGreen}{\textbf{ok}} \\
    \code{useradd deploy} & \textcolor{fitRed}{\textbf{lỗi}} {\scriptsize(user exists)}
      & \code{user: state=present} & \textcolor{fitGreen}{\textbf{ok}} \\
    \code{apt-get install nginx} & \textcolor{fitRed}{\textbf{luôn changed}}
      & \code{apt: state=present} & \textcolor{fitGreen}{\textbf{ok}} \\
  \end{fittable}
  \vspace{0.45cm}
  \begin{itemize}\setlength{\itemsep}{0.2cm}
    \item Shell: lần 2 \textbf{lỗi}, \textbf{nhân đôi cấu hình} hoặc báo \code{changed} sai.
    \item Module: lần 2 toàn \code{ok} $\rightarrow$ chạy lại bao nhiêu lần cũng \textbf{an toàn}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Khi buộc phải dùng shell}
  \noindent{\small Không có module phù hợp $\rightarrow$ \textbf{tự nói cho Ansible biết} khi nào lệnh đã làm rồi.}
  \par\vspace{0.3cm}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
\begin{codeblock}[minted options={fontsize=\scriptsize}]{yaml}
- name: Giải nén app
  ansible.builtin.shell: >
    tar -xzf app.tgz -C /opt/app
  args:
    creates: /opt/app/README

- name: Xem phiên bản kernel
  ansible.builtin.command: uname -r
  changed_when: false
\end{codeblock}
    \end{column}
    \begin{column}{0.42\textwidth}
      \raggedright
      \begin{fitnote}
        \code{creates} / \code{removes}\\
        File \textbf{đã có} (hoặc \textbf{đã mất}) $\rightarrow$ \textbf{bỏ qua} task,
        lần 2 báo \code{ok}.
      \end{fitnote}
      \vspace{0.2cm}
      \begin{fitnote}
        \code{changed\_when: false}\\
        Lệnh \textbf{chỉ đọc} $\rightarrow$ không bao giờ báo \code{changed}.
      \end{fitnote}
    \end{column}
  \end{columns}
  \vspace{0.3cm}
  {\small \textbf{Bổ sung}: \code{register} + \code{changed\_when}/\code{failed\_when}
   $\rightarrow$ tự đặt điều kiện theo output lệnh.}
\end{frame}

\begin{frame}{Hạn chế của Ansible}
  \begin{itemize}\setlength{\itemsep}{0.3cm}
    \item \textbf{Chậm khi nhiều máy}: mỗi task phải SSH vào và đẩy module sang;
          mặc định chỉ chạy song song \code{forks = 5} máy.
    \item \textbf{Không tự sửa khi lệch cấu hình}: không có agent nên ai sửa tay trên máy
          thì Ansible không biết $\rightarrow$ phải chạy lại playbook (hoặc lên lịch chạy định kỳ).
    \item \textbf{Playbook lớn khó đọc}: YAML + Jinja2 + \code{when}/\code{loop} dễ rối,
          debug khó hơn code thông thường.
    \item \textbf{Không quản lý vòng đời hạ tầng}: không lưu state như Terraform
          $\rightarrow$ không hợp để tạo/xoá VM hàng loạt.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Ví dụ 2: CI/CD — push code là chạy playbook}
  \centering
  \begin{tikzpicture}[
      st/.style={rounded corners=3pt, draw=fitBlue, thick, fill=fitLight, align=center,
        font=\scriptsize, minimum width=2.4cm, minimum height=0.85cm},
      ar/.style={-{Stealth[length=2mm]}, thick, fitBlue}]
    \node[st] (a) {\code{git push main}};
    \node[st, right=0.45cm of a] (b) {GitHub Actions\\khởi động runner};
    \node[st, right=0.45cm of b] (c) {\code{ansible-lint}\\kiểm tra};
    \node[st, right=0.45cm of c] (d) {\code{ansible-playbook}};
    \node[st, right=0.45cm of d, fill=fitBlue, text=white] (e) {\textbf{Server}\\\textbf{được cập nhật}};
    \draw[ar] (a) -- (b); \draw[ar] (b) -- (c); \draw[ar] (c) -- (d); \draw[ar] (d) -- (e);
  \end{tikzpicture}
  \par\vspace{0.2cm}
  \begin{columns}[t]
    \begin{column}{0.6\textwidth}
\begin{codeblock}[minted options={fontsize=\tiny}]{yaml}
# .github/workflows/deploy.yml
on:
  push:
    branches: [main]
jobs:
  deploy:
    runs-on: ubuntu-latest
    env:
      ANSIBLE_HOST_KEY_CHECKING: "False"
    steps:
      - uses: actions/checkout@v4
      - run: pip install ansible ansible-lint
      - run: ansible-lint site.yml
      - run: |
          echo "${{ secrets.SSH_KEY }}" > key && chmod 600 key
          echo "${{ secrets.VAULT_PASS }}" > .vault_pass
          ansible-playbook -i inventory site.yml \
            --private-key key --vault-password-file .vault_pass
\end{codeblock}
    \end{column}
    \begin{column}{0.38\textwidth}
      \raggedright
      \begin{itemize}\small\setlength{\itemsep}{0.25cm}
        \item Private key và mật khẩu Vault cất trong \textbf{GitHub Secrets},
              không nằm trong repo.
        \item \code{ansible-lint} lỗi $\rightarrow$ pipeline dừng, không đụng tới server.
        \item Nhờ \textbf{idempotent}: push 10 lần thì chỉ phần thay đổi mới báo
              \code{changed}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
